\begin{frame}{The Problem}
  \begin{itemize}
    \item Function calling is the bridge from LLMs to real-world tool use: APIs, agents, on-device assistants.
    \item Frontier models (GPT-4o, Claude) handle it well, but they are too large and too expensive to run locally.
    \item Sub-3B models trail by a wide margin on the Berkeley Function Calling Leaderboard (BFCL).
    \item The gap matters most where it is hardest to close: phones, laptops, free-tier Colab, classrooms.
  \end{itemize}
  \vspace{6pt}
  {\small\color{dimgray} Goal: push Qwen2.5-3B-Instruct (Apache 2.0) closer to the frontier on BFCL V4 without crossing the 3B parameter line.}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Approach}
  \textbf{Three levers, all applied to Qwen2.5-3B-Instruct.}
  \begin{itemize}
    \item \textbf{Cold-start GRPO.} Skip SFT warm-up before RL. Following ToolRL\footnote{\href{https://arxiv.org/abs/2504.13958}{arXiv:2504.13958}}, supervised warm-up at small scale memorizes traces and limits exploration.
    \item \textbf{Hammer-style function masking.} Replace 67\% of function and parameter names with random identifiers so the model learns to read tool descriptions, not memorize names.\footnote{\href{https://arxiv.org/abs/2410.04587}{Hammer 2.1, arXiv:2410.04587}}
    \item \textbf{Single-call to parallel-call curriculum.} Stage easier examples first, scale to harder.
  \end{itemize}
  \vspace{4pt}
  {\small\color{dimgray} Stack: QLoRA rank 64 + NF4 quantization, paged AdamW 8-bit, TRL SFTTrainer / GRPOTrainer.}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Data: xLAM-60k to a 4K Curated Subset}
  \textbf{Source.} Salesforce \texttt{xLAM-function-calling-60k} (CC-BY-4.0)\footnote{\href{https://huggingface.co/datasets/Salesforce/xlam-function-calling-60k}{HF: Salesforce/xlam-function-calling-60k}}: 60{,}264 query / tool-set / call triples. Roughly 47\% single-call, 53\% parallel-call, 0\% no-call.
  \vspace{4pt}

  \textbf{Four-stage filter to 4{,}000 examples:}
  \begin{enumerate}
    \item \textbf{Validation.} Drop malformed JSON, missing fields, calls referencing unknown tools.
    \item \textbf{Deduplication.} Case-insensitive exact-match dedupe on query text.
    \item \textbf{Balanced sampling.} Target 60 / 30 / 10 (single / parallel / no-call). xLAM has zero no-call, so final mix is 64.7\% single, 35.3\% parallel.
    \item \textbf{Diversity scoring.} Within each bucket, rank by parameter-type diversity (string, int, float, bool, array).
  \end{enumerate}
  \vspace{4pt}
  {\small\color{dimgray} The empty no-call slot is a known limitation; see slide on Limitations.}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}[fragile]{Function Masking: Before / After}
  Mask 67\% of training examples (2{,}680 masked / 1{,}320 unmasked). Function names, parameter names, and default values get random identifiers.
  \vspace{4pt}

  \textbf{Before (raw xLAM):}
\begin{lstlisting}
{
  "name": "get_weather",
  "parameters": {
    "city": {"type": "string"},
    "units": {"type": "string", "default": "celsius"}
  }
}
\end{lstlisting}

  \textbf{After (masked):}
\begin{lstlisting}
{
  "name": "k7q3xz",
  "parameters": {
    "p2vrm9": {"type": "string"},
    "ab8ftc": {"type": "string", "default": "rW82"}
  }
}
\end{lstlisting}

  {\small\color{dimgray} The model is forced to read the natural-language tool description, not pattern-match on \texttt{get\_weather}.}
\end{frame}

\begin{frame}{Systems Contributions: Three bfcl-eval Patches}
  Validating end-to-end on Colab T4 surfaced three reproducibility bugs in upstream \texttt{bfcl-eval}. All fixed in our fork\footnote{\href{https://github.com/arnursabet/gorilla/tree/qwen-2.5-3b-handler-t4-support}{arnursabet/gorilla @ qwen-2.5-3b-handler-t4-support, commit \texttt{f64471b}}}.
  \vspace{4pt}
  \begin{enumerate}
    \item \textbf{Silent dtype kwarg.} \texttt{QwenFCHandler.\_\_init\_\_} declared a \texttt{dtype} kwarg but never forwarded it to \texttt{super().\_\_init\_\_()}. Parent's \texttt{bfloat16} default always won; T4 (CC 7.5) has no bf16 support, so every run silently failed. Fix: forward the kwarg.
    \item \textbf{Hardcoded vLLM args.} \texttt{base\_oss\_handler} built \texttt{vllm serve} with no knob for \texttt{--max-num-seqs} / \texttt{--max-model-len}. Default \texttt{max\_num\_seqs=256} OOMs on a 15\,GB T4. Fix: env vars (\texttt{BFCL\_VLLM\_MAX\_NUM\_SEQS}, \texttt{BFCL\_VLLM\_MAX\_MODEL\_LEN}).
    \item \textbf{Context-length desync.} \texttt{leftover\_tokens\_count} was capped at 4{,}096, oblivious to the actual server \texttt{--max-model-len}. Fix: shrink \texttt{self.max\_context\_length} to match the env var when set.
  \end{enumerate}
  \vspace{2pt}
  {\small\color{dimgray} Non-breaking on A100/H100. Lowers the bar for free-tier classroom replication.}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Limitations and Future Work}
  \textbf{Limitations.}
  \begin{itemize}
    \item \textbf{Training set size.} 4K examples is roughly an order of magnitude smaller than xLAM-2 / Hammer 2.1. Likely underfits the long tail.
    \item \textbf{Adapter capacity.} Rank-64 QLoRA on NF4 keeps trainable params under 5\%. Caps expressivity, especially on Multi-Turn.
    \item \textbf{No-call synthesis did not ship.} 400-example refusal-training stage was scoped but unimplemented. Hallucination gains depend on emergent generalization.
    \item \textbf{Agentic bucket may be incomplete.} Requires Serper API credits. We may report Non-Live, Live, Hallucination, Multi-Turn separately if Agentic is missing.
  \end{itemize}

  \textbf{Future work.} No-call synthesis, then a rank-128 ablation, then a budgeted Agentic run for fully comparable V4 overall.
\end{frame}
